<script>

const canvas = document.getElementById('map');

const ctx = canvas.getContext('2d');

const logEl = document.getElementById('log');


const COLS = 16;

const ROWS = 12;

const CELL = 40;


let state = null;


function log(msg, cls = '') {

  const d = document.createElement('div');

  if (cls) d.className = cls;

  d.innerHTML = msg;

  logEl.appendChild(d);

  logEl.scrollTop = logEl.scrollHeight;

}


function createInitialState() {

  const terrain = [];

  for (let y = 0; y < ROWS; y++) {

    terrain[y] = [];

    for (let x = 0; x < COLS; x++) {

      terrain[y][x] = 0; //0平地 1山 2丛林

    }

  }

  // 阻挡地形

  const obstacles = [

    [4,3],[5,3],[4,4],[11,2],[12,2],[11,3],

    [3,8],[4,8],[4,9],[10,8],[11,8],[12,9],

    [7,5],[8,5],[8,6],[7,7]

  ];

  obstacles.forEach(([x,y]) => {

    terrain[y][x] = Math.random() < 0.5 ? 1 : 2;

  });


  return {

    terrain,

    fog: Array.from({length: ROWS}, () => Array(COLS).fill(true)),

    units: [

      { id:1, team:'blue', type:'artillery', x:1, y:2, hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },

      { id:2, team:'blue', type:'antiair',  x:1, y:5, hp:8,  maxHp:8,  move:4, range:3, attack:3, moved:false, fired:false },

      { id:3, team:'blue', type:'artillery', x:2, y:4, hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },

      { id:4, team:'blue', type:'antiair',  x:2, y:7, hp:8,  maxHp:8,  move:4, range:3, attack:3, moved:false, fired:false },

      { id:5, team:'blue', type:'artillery', x:1, y:9, hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },


      { id:11, team:'red', type:'artillery', x:14, y:1, hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },

      { id:12, team:'red', type:'antiair',  x:14, y:4, hp:8,  maxHp:8,  move:4, range:3, attack:3, moved:false, fired:false },

      { id:13, team:'red', type:'artillery', x:13, y:6, hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },

      { id:14, team:'red', type:'antiair',  x:14, y:9, hp:8,  maxHp:8,  move:4, range:3, attack:3, moved:false, fired:false },

      { id:15, team:'red', type:'artillery', x:13, y:10,hp:10, maxHp:10, move:3, range:4, attack:4, moved:false, fired:false },

    ],

    selectedId: null,

    turn: 'blue',

    turnNumber: 1

  };

}


function isPassable(x, y) {

  if (x < 0 || y < 0 || x >= COLS || y >= ROWS) return false;

  const t = state.terrain[y][x];

  return t !== 1 && t !== 2;

}


function unitAt(x, y) {

  return state.units.find(u => u.x === x && u.y === y && u.hp > 0);

}


// 计算我方视野，更新迷雾

function updateFog() {

  state.fog = Array.from({length: ROWS}, () => Array(COLS).fill(true));

  state.units.forEach(u => {

    if (u.hp > 0 && u.team === 'blue') {

      revealFog(u.x, u.y, 2);

    }

  });

}

function revealFog(x, y, radius = 2) {

  for (let dy = -radius; dy <= radius; dy++) {

    for (let dx = -radius; dx <= radius; dx++) {

      const nx = x + dx, ny = y + dy;

      if (nx >= 0 && ny >= 0 && nx < COLS && ny < ROWS) {

        if (Math.abs(dx) + Math.abs(dy) <= radius + 1) {

          state.fog[ny][nx] = false;

        }

      }

    }

  }

}


function drawCell(x, y) {

  const px = x * CELL, py = y * CELL;

  const t = state.terrain[y][x];


  if (t === 1) {

    ctx.fillStyle = '#5a5f66';

    ctx.fillRect(px, py, CELL, CELL);

    ctx.strokeStyle = '#3b3f45';

    ctx.strokeRect(px, py, CELL, CELL);

    ctx.fillStyle = '#7c828a';

    ctx.beginPath();

    ctx.moveTo(px + 6, py + 32);

    ctx.lineTo(px + 20, py + 8);

    ctx.lineTo(px + 34, py + 32);

    ctx.closePath();

    ctx.fill();

  } else if (t === 2) {

    ctx.fillStyle = '#2f4a2f';

    ctx.fillRect(px, py, CELL, CELL);

    ctx.strokeStyle = '#1f3520';

    ctx.strokeRect(px, py, CELL, CELL);

    ctx.fillStyle = '#4caf50';

    for (let i = 0; i < 5; i++) {

      ctx.beginPath();

      ctx.arc(px + 8 + i * 6, py + 10 + (i % 2) * 18, 3, 0, Math.PI * 2);

      ctx.fill();

    }

  } else {

    ctx.fillStyle = '#c8b98a';

    ctx.fillRect(px, py, CELL, CELL);

    ctx.strokeStyle = '#a89a6b';

    ctx.strokeRect(px, py, CELL, CELL);

  }


  if (state.fog[y][x]) {

    ctx.fillStyle = 'rgba(10, 14, 20, 0.82)';

    ctx.fillRect(px, py, CELL, CELL);

  }

}


function drawUnit(u) {

  if (u.hp <= 0) return;

  // 关键：红方单位，不在我方视野就不绘制！

  if (u.team === 'red' && state.fog[u.y][u.x]) return;


  const px = u.x * CELL, py = u.y * CELL;

  const cx = px + CELL / 2, cy = py + CELL / 2;


  if (u.team === 'blue') {

    ctx.fillStyle = '#3e8ef7';

    ctx.strokeStyle = '#cfe4ff';

  } else {

    ctx.fillStyle = '#e55555';

    ctx.strokeStyle = '#ffd1d1';

  }


  if (u.type === 'artillery') {

    ctx.fillRect(cx - 10, cy - 4, 20, 8);

    ctx.fillStyle = ctx.strokeStyle;

    ctx.fillRect(cx + 4, cy - 2, 8, 4);

    ctx.fillStyle = u.team === 'blue' ? '#3e8ef7' : '#e55555';

    ctx.beginPath();

    ctx.arc(cx - 8, cy + 4, 4, 0, Math.PI * 2);

    ctx.arc(cx + 8, cy + 4, 4, 0, Math.PI * 2);

    ctx.fill();

  } else {

    ctx.fillRect(cx - 10, cy - 6, 20, 12);

    ctx.fillStyle = ctx.strokeStyle;

    ctx.fillRect(cx - 4, cy - 10, 8, 4);

    ctx.fillStyle = u.team === 'blue' ? '#3e8ef7' : '#e55555';

    ctx.beginPath();

    ctx.arc(cx - 8, cy + 6, 3, 0, Math.PI * 2);

    ctx.arc(cx + 8, cy + 6, 3, 0, Math.PI * 2);

    ctx.fill();

  }


  const hpRatio = u.hp / u.maxHp;

  ctx.fillStyle = '#000';

  ctx.fillRect(px + 4, py + 2, CELL - 8, 4);

  ctx.fillStyle = hpRatio > 0.5 ? '#76ff9a' : hpRatio > 0.25 ? '#ffd166' : '#ff6b6b';

  ctx.fillRect(px + 4, py + 2, (CELL - 8) * hpRatio, 4);


  if (u.id === state.selectedId) {

    ctx.strokeStyle = '#ffe066';

    ctx.lineWidth = 2;

    ctx.strokeRect(px + 1, py + 1, CELL - 2, CELL - 2);

    ctx.lineWidth = 1;

  }

}


function drawMoveRange(unit) {

  if (!unit || unit.moved) return;

  for (let y = 0; y < ROWS; y++) {

    for (let x = 0; x < COLS; x++) {

      if (!isPassable(x, y)) continue;

      if (unitAt(x, y)) continue;

      const dist = Math.abs(x - unit.x) + Math.abs(y - unit.y);

      if (dist > 0 && dist <= unit.move) {

        ctx.fillStyle = 'rgba(118, 255, 154, 0.22)';

        ctx.fillRect(x * CELL, y * CELL, CELL, CELL);

        ctx.strokeStyle = 'rgba(118, 255, 154, 0.6)';

        ctx.strokeRect(x * CELL + 1, y * CELL + 1, CELL - 2, CELL - 2);

      }

    }

  }

}


function drawAttackRange(unit) {

  if (!unit || unit.fired) return;

  for (let y = 0; y < ROWS; y++) {

    for (let x = 0; x < COLS; x++) {

      const dist = Math.abs(x - unit.x) + Math.abs(y - unit.y);

      if (dist > 0 && dist <= unit.range) {

        ctx.strokeStyle = 'rgba(255, 107, 107, 0.7)';

        ctx.setLineDash([4, 4]);

        ctx.strokeRect(x * CELL + 2, y * CELL + 2, CELL - 4, CELL - 4);

        ctx.setLineDash([]);

      }

    }

  }

}


function render() {

  ctx.clearRect(0, 0, canvas.width, canvas.height);

  for (let y = 0; y < ROWS; y++) {

    for (let x = 0; x < COLS; x++) {

      drawCell(x, y);

    }

  }

  const selected = state.units.find(u => u.id === state.selectedId);

  drawMoveRange(selected);

  drawAttackRange(selected);

  state.units.forEach(u => drawUnit(u));

}


function clickCell(x, y) {

  if (state.turn !== 'blue') return;

  const clickedUnit = unitAt(x, y);


  if (clickedUnit && clickedUnit.team === 'blue') {

    state.selectedId = clickedUnit.id;

    log(`选中我方 ${clickedUnit.type === 'artillery' ? '火炮' : '防空车'}`);

    render();

    return;

  }


  if (state.selectedId) {

    const unit = state.units.find(u => u.id === state.selectedId);

    if (!unit) return;


    if (clickedUnit && clickedUnit.team === 'red') {

      const dist = Math.abs(x - unit.x) + Math.abs(y - unit.y);

      if (dist <= unit.range && !unit.fired) {

        const dmg = unit.attack;

        clickedUnit.hp -= dmg;

        unit.fired = true;

        log(`${unit.type === 'artillery' ? '火炮' : '防空车'}攻击红方，造成 ${dmg} 伤害`);

        if (clickedUnit.hp <= 0) log(`红方单位被歼灭！`);

        checkVictory();

        render();

        return;

      } else {

        log(unit.fired ? '本单位已经开火过' : '目标超出射程');

        return;

      }

    }


    if (isPassable(x, y) && !unitAt(x, y)) {

      const dist = Math.abs(x - unit.x) + Math.abs(y - unit.y);

      if (dist <= unit.move && !unit.moved) {

        unit.x = x;

        unit.y = y;

        unit.moved = true;

        updateFog();

        log(`${unit.type === 'artillery' ? '火炮' : '防空车'}移动完成，现在可以开火！`);

        render();

        return;

      } else {

        log(unit.moved ? '本单位已经移动完毕' : '超出移动距离');

        return;

      }

    }

  }

}


async function enemyTurn() {

  state.turn = 'red';

  log(`—— 红方回合 ——`);


  const reds = state.units.filter(u => u.team === 'red' && u.hp > 0);

  const blues = state.units.filter(u => u.team === 'blue' && u.hp > 0);


  for (const u of reds) {

    if (blues.length === 0) break;

    let target = null, bestDist = Infinity;

    blues.forEach(b => {

      const d = Math.abs(b.x - u.x) + Math.abs(b.y - u.y);

      if (d < bestDist) { bestDist = d; target = b; }

    });

    if (!target) continue;


    if (bestDist <= u.range) {

      // 敌方攻击，但是敌人可能在迷雾里，只提示受伤，不显示敌人

      target.hp -= u.attack;

      log(`<span class="warn">警告！我方单位遭到敌方炮火打击，受到 ${u.attack} 伤害（看不到攻击者）</span>`);

      if (target.hp <= 0) log(`我方单位被击毁`, 'red');

    } else {

      const dx = Math.sign(target.x - u.x);

      const dy = Math.sign(target.y - u.y);

      let moved = false;

      for (let s = u.move; s > 0 && !moved; s--) {

        const nx = u.x + dx * s, ny = u.y + dy * s;

        if (isPassable(nx, u.y) && !unitAt(nx, u.y)) { u.x = nx; moved = true; }

        else if (isPassable(u.x, ny) && !unitAt(u.x, ny)) { u.y = ny; moved = true; }

      }

      if (moved) log(`红方单位向前推进`);

    }

    checkVictory();

    render();

    await sleep(400);

  }


  state.units.forEach(u => { u.moved = false; u.fired = false; });

  state.turn = 'blue';

  state.turnNumber++;

  log(`—— 第 ${state.turnNumber} 回合 · 我方行动 ——`);

  render();

}

function sleep(ms){return new Promise(r=>setTimeout(r,ms));}


function checkVictory() {

  const blues = state.units.filter(u => u.team === 'blue' && u.hp > 0);

  const reds = state.units.filter(u => u.team === 'red' && u.hp > 0);

  if (reds.length === 0) {

    log(`🎉 胜利：全歼红方！`);

    state.turn = 'blue';

  } else if (blues.length === 0) {

    log(`💀 失败：我方全灭`);

    state.turn = 'red';

  }

}


canvas.addEventListener('click', e => {

  const rect = canvas.getBoundingClientRect();

  const x = Math.floor((e.clientX - rect.left) / CELL);

  const y = Math.floor((e.clientY - rect.top) / CELL);

  clickCell(x, y);

});


document.getElementById('endTurn').addEventListener('click', async () => {

  if (state.turn !== 'blue') return;

  state.selectedId = null;

  await enemyTurn();

});


document.getElementById('reset').addEventListener('click', () => {

  state = createInitialState();

  updateFog();

  logEl.innerHTML = '';

  log(`新关卡开始：选中火炮或防空车，移动后可以直接攻击敌人`);

  render();

});


state = createInitialState();

updateFog();

log(`开始：选中单位移动，移动完成后可攻击。小心！敌人能在迷雾里偷袭你！`);

render();

</script>
